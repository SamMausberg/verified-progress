% Evidence status. A committed result cites its entry in the evidence register
% (appendix, \label{ev:<key>}): \ev{key} or \ev{key1,key2} prints [E3] or [E3, E7]. A result
% that is designed but not committed carries the single pending marker \pend{key}, which
% prints [pending D4] and links to the pending list (\label{dev:<key>}); the prose beside
% it says what the result will decide, and no number stands in for it.
\newcommand{\evref}[1]{\hyperref[ev:#1]{E\ref*{ev:#1}}}
\newcommand{\evloop}[1]{\evsep\evref{#1}\def\evsep{, }}
\DeclareRobustCommand{\ev}[1]{{\def\evsep{}[\forcsvlist{\evloop}{#1}]}}
\newcommand{\devref}[1]{\hyperref[dev:#1]{D\ref*{dev:#1}}}
\newcommand{\devloop}[1]{\devsep\devref{#1}\def\devsep{, }}
\DeclareRobustCommand{\pend}[1]{\mbox{{\def\devsep{}[pending~\forcsvlist{\devloop}{#1}]}}}
% A figure that reads a committed data file: draw it if the file exists, otherwise a
% pending box naming the file, so the source builds before the evidence merges.
\newcommand{\pendingfigure}[2]{%
  \begin{center}\setlength{\fboxsep}{6pt}%
  \fbox{\parbox{.92\linewidth}{\centering\small Pending figure. #1\par\smallskip
  \footnotesize Data: \texttt{\detokenize{#2}}}}\end{center}}
\newcommand{\datafigure}[3]{\IfFileExists{#1}{#2}{\pendingfigure{#3}{#1}}}
