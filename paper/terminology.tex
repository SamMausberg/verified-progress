% Terminology, declared once (macros.tex explains \term). Scope main: terms the main text
% relies on (owner: main-text author). Scope app: terms used only in the appendices (owner:
% appendix editor). The glossary (Appendix app:glossary) prints both scopes.
\term{main}{Vocabulary entry $i$}{A token of the vocabulary and its row $w_i$ of the head matrix $W$; the rows of $W$ are always called entries, never positions.}
\term{main}{Position}{One head input $h$ and the token decided from it: a decode step's request, or one drafted token under verification.}
\term{main}{Head input $h$}{The BF16 vector the engine feeds to the output head (the final norm's output) at one position.}
\term{main}{Batch shape $M$}{The number of positions in one head call; cuBLAS chooses its kernel by it. $B$ is the number of requests in a decode step.}
\term{main}{Real logit $z_i$}{$\langle w_i,h\rangle$ in exact arithmetic over the stored BF16 operands.}
\term{main}{Accumulator $\hat y_i$}{The FP32 sum a kernel forms for a logit before rounding it to BF16.}
\term{main}{Stock head}{SGLang's head at the pinned commit: a cuBLAS BF16 GEMM with FP32 accumulation and BF16 output, then \code{torch.argmax} (first maximal index).}
\term{main}{\Rreal, \Rbf, \Rstock}{The three references for one greedy decision (Definition~\ref{def:contracts}).}
\term{main}{Error model $\gamma_{\rm stock}$, $G_i$}{An assumed bound $G_i=\gamma_{\rm stock}\sum_j|w_{ij}h_j|+\eta$ on the stock kernel's accumulation error, with $\gamma_{\rm stock}$ from the \modelcons{} or the \modelhopper{} (Section~\ref{sec:headcontracts}, Table~\ref{tab:assumptions}).}
\term{main}{Error bound $\beta_i$}{A proved bound on the error of the int8 copy's score for entry $i$ (Theorem~\ref{thm:envelope}).}
\term{main}{Interval $[lo_i,hi_i]$}{An interval that contains the value it bounds, given the stated assumptions, computed with outward rounding; the appendices call it an enclosure.}
\term{main}{Candidate}{A vocabulary entry the screen cannot exclude; it is recomputed from its BF16 row.}
\term{main}{Certified position}{A position whose \Rstock{} token the intervals fix for every accumulator value the error model admits.}
\term{main}{Fallback}{The stock head at the same batch shape, for a position the intervals cannot decide: on the whole batch, or per position over its gathered candidates, the per-position fallback. The evidence files call the per-position fallback the column fallback.}
\term{main}{Capture, replay}{Head inputs recorded bit for bit from the running engine; their offline re-evaluation (Section~\ref{sec:transport-capture}).}
\term{main}{Held-out}{From prompts that no choice of head, bound or threshold was fitted on.}
\term{main}{Transport; tile summary}{Bounds carried from a drafter's per-tile summaries to the target's head input; the largest score and the sum of exponentiated scores of a tile of vocabulary entries at the drafter's head input.}
\term{main}{$\rho$; $\delta^\ast_i$}{Drift $\norm{h^t-h^d}_2/\norm{h^t}_2$ of a draft pair; the $\rho$ below which transport's bound beats the int8 copy's on entry $i$.}

% Scope app (owner: appendix editor): terms used only in the appendices, in alphabetical
% order; printed by the glossary (Appendix app:glossary).
\term{app}{Accepted length}{Committed tokens per verify cycle, the bonus token included (Table~\ref{tab:profile-mtp}).}
\term{main}{Batch size $B$, concurrency $c$}{$B$ is the number of requests in one decode step; $c$ is the number of requests the client keeps in flight, which the server may cap below $c$.}
\term{app}{Batch union}{The set of vocabulary tiles or entries that at least one position of a batch still needs; a kernel that shares weight reads across the batch must read the whole union (Appendix~\ref{app:transport-sampled}).}
\term{main}{Bound family, tiling}{The geometry of a transport bound (scalar, coordinate, grouped, low-rank radii, or their minimum) and the partition of the vocabulary into tiles (contiguous, random or $k$-means; 64 to 1,024 entries) (Appendix~\ref{app:transport-geometry}).}
\term{app}{Commit rule, sound set}{Commit a decision when every logit vector in a set known to contain the reference logits gives the same decision (Definition~\ref{def:commit}).}
\term{main}{Divergence, exposure, noise floor}{The first position at which two runs choose different tokens; the positions compared before it; divergences per 1,000 exposed tokens between stock configurations that ought to agree (Appendix~\ref{app:divergence}).}
\term{main}{Drift scaling $s$}{A model extrapolation that replaces the draft input by $h^t-s\Delta$, scaling every drift term by $s$; not a measured drafter (Appendix~\ref{app:transport-scaling}).}
\term{app}{First differing module output}{In a bit-exact tap of two runs, the first position and, in execution order, the first module whose output bits differ; cache state is checked separately (Appendix~\ref{app:divergence}).}
\term{main}{Fixed-noise verification}{Sampled verification that accepts a drafted token exactly when it wins the target's race under the keyed noise field (Theorem~\ref{thm:fixednoise}).}
\term{app}{Frontier coordinates $x$, $y$}{Mean per-request output rate including the time to first token, and output tokens per second on the GPU (Equation~\eqref{eq:frontier}).}
\term{main}{Guards, $P_?$}{Interval tests that decide the standard acceptance rule from enclosures of the numerator and the partition function; $P_?$ is the probability that neither fires (Theorem~\ref{thm:guards}).}
\term{app}{Head chain}{The stock head GEMM, the FP32 copy of its logits and the argmax or top-1 that follows it (Appendix~\ref{app:profile}).}
\term{main}{Integer pass}{A certified pass that quantizes the head input as well and accumulates exactly in int32 (Theorem~\ref{thm:integer}).}
\term{main}{Keyed noise field}{Gumbel noise $G_{t,i}$ that is a pure function of (seed, token position $t$ in the sequence, token $i$), as in the engine's seeded sampler.}
\term{main}{Oracle radii}{The realized per-tile deviations used as transport radii; a ceiling that no certified geometry reaches (Appendix~\ref{app:transport-families}).}
\term{app}{Partition interval $\log(Z^+/Z^-)$}{The width, in nats, of an enclosure of the partition function; it decides how often the acceptance guards fire (Appendix~\ref{app:transport-sampled}).}
\term{app}{Point-mass proposal}{A deterministic draft token, SGLang's default for sampled verification; its residual is a race over the untried tokens (Appendix~\ref{app:residual}).}
\term{main}{Residual race}{An exact sample from the residual law $(p-q)_+$ by a certified race under a fixed priority field (Theorem~\ref{thm:residualrace}).}
\term{main}{Static screen}{Tile bounds $\langle\mu_c,h\rangle+r_c\norm h$ evaluated at the target state without any draft (Proposition~\ref{prop:static}).}
\term{app}{\Srace}{The reference for one seeded decision: the winner of the race with a fixed noise vector (Definition~\ref{def:srace}).}
\term{app}{Tuning and confirmation splits}{Disjoint prompt sets of the serving workload: configurations are chosen on the first and reported on the second (Appendix~\ref{app:serving}).}
\term{main}{Verify pass, draft step, cycle}{The target's forward pass over a drafted block; one drafter forward; a draft, its verification and the state commit (Appendix~\ref{app:profile}).}
