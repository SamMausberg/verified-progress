% Appendices. paper.tex loads this file after the bibliography,
% under \appendix. The appendices are set in one column: several of their tables are
% longtables, which two-column mode does not support, or are wider than a column.
\onecolumn
% A page that holds only floats would otherwise centre them and spread them apart with
% stretchable glue (LaTeX's default \@fptop, \@fpsep and \@fpbot). Stack them from the
% top at the usual float separation instead; the main text's two-column float pages
% (\@dblfp...) are not affected.
\makeatletter
\setlength{\@fptop}{0pt}
\setlength{\@fpsep}{\floatsep}
\setlength{\@fpbot}{0pt plus 1fil}
% A page of floats alone must be at least three quarters full; a smaller float goes to the top of a
% page with text below it instead of leaving half a page empty.
\renewcommand{\floatpagefraction}{0.75}
\makeatother
\usetikzlibrary{shapes.geometric}
% The style sets \flushbottom with a rigid \parskip, so a paragraph that cannot be split at the
% foot of a page leaves a blank band there. A little stretch between paragraphs lets the page
% fill instead.
\setlength{\parskip}{6pt plus 2pt}
% Start a new page unless at least #1 of space is left, so that no appendix heading sits at the
% foot of a page with a few lines under it (the needspace package's test, without the package).
\makeatletter
\newcommand{\appneedspace}[1]{\par\penalty-100\begingroup\setlength{\dimen@}{#1}%
  \dimen@ii\pagegoal\advance\dimen@ii-\pagetotal
  \ifdim\dimen@>\dimen@ii\ifdim\dimen@ii>\z@\vfil\fi\penalty-\@M\fi\endgroup}
\makeatother
