% Figure fig:breakdown: where the GPU time of a decode step goes. The bars read
% evidence/profiles/breakdown.csv (one row per configuration, columns in percent of the
% profiled step or cycle; the components sum to 100). Rows 0-3 of the file are plain
% decoding at B = 1, 8, 32, 128 and rows 4-6 the MTP cycle at B = 1, 8, 32 (column config);
% the plot leaves 1.5 extra rows between the two groups (\vprow). A third group, one bar, is the DFlash
% cycle of the tuned arm with 16-token blocks at B = 1: the first row of
% evidence/profiles/dflash_breakdown.csv (config dflash-tuned-b16-1), appended to the table below
% (the plots skip that file's other rows) after another 1.5 extra rows. Its config name has hyphens, so its arm and batch for the step_share.csv
% lookup are set by hand, and its tick label is written out. Its idle is not hatched: that arm is
% GPU-bound and tracing does not inflate its idle (evidence/profiles/README.md).
% Components: head = target head GEMM, FP32 logits copy and argmax; draft_head = the draft
% steps' head GEMMs, copies and top-1; weight_gemms = the target's weight GEMMs ("target
% GEMMs"); "other kernels" = gdn_state (the Gated DeltaNet recurrence and the other non-GEMM
% GDN kernels), attention, small kernels and draft_model (the MTP layer's non-head work, its
% GEMMs included) (evidence/profiles/README.md). The head shares printed at the right are
% (head + draft_head) / step from the microseconds of evidence/profiles/step_share.csv,
% which round as the README's table does (the rounded percentages of breakdown.csv would
% give 12.5% instead of 12.4% for MTP at B = 32).
% Colours follow style.tex: the target's head in ink (the stock engine), each drafter's heads in its
% arm's colour (MTP green in the MTP rows, 16-token DFlash reddish purple in the DFlash row), the
% target GEMMs in a 40% grey and every other kernel in a 12% grey, GPU idle as an empty white
% segment. The fills stay apart in a greyscale print (grey levels about 56, 86, 147, 200, 239
% and 255 of 255 for ink, green, reddish purple, the two greys and white). The MTP idle
% segments are hatched because tracing inflates them, which the caption says. The column at the right gives the target head's share
% and, for the speculative rows, the drafter's heads' share separately (target + draft), each from
% the microseconds of step_share.csv.
% Segments are labelled directly, no legend: the target head, target GEMMs and other kernels
% above the B = 1 plain bar, with a leader to the segment where the label is not over it; the
% draft heads and GPU idle in the gap between the plain and MTP groups (GPU idle with leaders up
% to plain B = 128's idle and down to MTP B = 1's hatched idle); DFlash's draft heads in the gap
% above the DFlash bar. The gaps are 2.5 rows, so a 7 pt label clears the bar above it by 2 pt and
% leaves room for a visible leader to the bar below; labels inside the plot are set on white to
% hide the grid.
\vpreadcsv{\breakdowntable}{../evidence/profiles/breakdown.csv}
\vpreadcsv{\dflashrow}{../evidence/profiles/dflash_breakdown.csv}
\pgfplotstablevertcat{\breakdowntable}{\dflashrow}
\vpreadcsv{\stepsharetable}{../evidence/profiles/step_share.csv}
\def\vpsplitconfig#1-#2\relax{\gdef\vparm{#1}\gdef\vpbatchsize{#2}}
\foreach \row in {0,...,7}{%
  \ifnum\row=7 \gdef\vparm{dflash-tuned-b16}\gdef\vpbatchsize{1}\else
  \pgfplotstablegetelem{\row}{config}\of\breakdowntable
  \expandafter\vpsplitconfig\pgfplotsretval\relax\fi
  \vplookupthree{\stepsharetable}{arm}{\vparm}{batch}{\vpbatchsize}{component}{head}{us_per_step}{\vphead}%
  \vplookupthree{\stepsharetable}{arm}{\vparm}{batch}{\vpbatchsize}{component}{draft_head}{us_per_step}{\vpdraft}%
  \vplookupthree{\stepsharetable}{arm}{\vparm}{batch}{\vpbatchsize}{component}{head}{step_us}{\vpstep}%
  \pgfmathsetmacro{\vpshare}{\vphead/\vpstep*100}%
  \pgfmathsetmacro{\vpdshare}{\vpdraft/\vpstep*100}%
  \global\expandafter\let\csname vpheadshare\row\endcsname\vpshare
  \global\expandafter\let\csname vpdraftshare\row\endcsname\vpdshare}
\def\vpbatch#1-#2\relax{$B{=}#2$}
\def\vppc#1{\pgfmathprintnumber[fixed,precision=1,fixed zerofill]{#1}}
\def\vprow#1{#1+1.5*(#1>3)+1.5*(#1>6)}
\begin{tikzpicture}
\begin{axis}[xbar stacked,y dir=reverse,scale only axis,width=5.2cm,height=3.36cm,
  bar width=6.5pt,xmin=0,xmax=100,ymin=-0.6,ymax=10.6,
  xlabel={Share of the profiled step or cycle},xlabel style={yshift=2pt},
  xtick={0,20,40,60,80,100},xticklabels={0\%,20\%,40\%,60\%,80\%,100\%},
  ytick=data,yticklabel={\ifnum\ticknum=7 DFlash, $B{=}1$\else\pgfplotstablegetelem{\ticknum}{config}\of\breakdowntable
    \expandafter\vpbatch\pgfplotsretval\relax\fi},
  major y tick style={draw=none},xmajorgrids,clip=false,
  every axis plot/.append style={draw=white,line width=.4pt,skip coords between index={8}{100}}]
\addplot[fill=cstock] table[x=head,y expr={\vprow{\coordindex}}]{\breakdowntable};
\addplot[fill=cmtp] table[x expr={(\coordindex<7)*\thisrow{draft_head}},y expr={\vprow{\coordindex}}]{\breakdowntable};
\addplot[fill=cdflash16] table[x expr={(\coordindex>6)*\thisrow{draft_head}},y expr={\vprow{\coordindex}}]{\breakdowntable};
\addplot[fill=ccontext!40] table[x=weight_gemms,y expr={\vprow{\coordindex}}]{\breakdowntable};
\addplot[fill=ccontext!12] table[x expr={\thisrow{gdn_state}+\thisrow{attention}+\thisrow{small_kernels}+\thisrow{draft_model}},y expr={\vprow{\coordindex}}]{\breakdowntable};
\addplot[fill=white,draw=ccontext] table[x expr={(\coordindex>3)*(\coordindex<7)?0:\thisrow{idle}},y expr={\vprow{\coordindex}}]{\breakdowntable};
\addplot[fill=white,draw=ccontext,pattern=north east lines,pattern color=ccontext] table[x expr={(\coordindex>3)*(\coordindex<7)?\thisrow{idle}:0},y expr={\vprow{\coordindex}}]{\breakdowntable};
% head shares at the right: target alone for plain decoding, target + draft for MTP and DFlash
\pgfplotsinvokeforeach{0,...,3}{%
  \node[annot,anchor=west] at (axis cs:100.5,#1)
    {\vppc{\csname vpheadshare#1\endcsname}\%};}
\pgfplotsinvokeforeach{4,...,7}{%
  \node[annot,anchor=west] at (axis cs:100.5,{\vprow{#1}})
    {\vppc{\csname vpheadshare#1\endcsname}${}+{}$\vppc{\csname vpdraftshare#1\endcsname}\%};}
\node[annot,anchor=south west,align=left] at (axis cs:100.5,-0.6) {heads:\\target + draft};
% Segment labels. A bar is 6.5 pt = 0.76 rows thick and a 7 pt label about 1 row (0.3 cm per
% row), so a label centred at row r ends near r + 0.5, where its leader starts. Above the plain B = 1 bar (top edge at
% -0.38), outside the plot: leaders run from each label down to its segment.
\node[annot,anchor=south west] at (axis cs:0,-0.9) {target head};
\node[annot,anchor=south] at (axis cs:44,-0.9) {target GEMMs};
\node[annot,anchor=south] at (axis cs:85,-0.9) {other kernels};
\draw[leader] (axis cs:5.1,-0.9) -- (axis cs:5.1,-0.38)
  (axis cs:44,-0.9) -- (axis cs:44,-0.38) (axis cs:85,-0.9) -- (axis cs:85,-0.38);
% In the gap between the plain and MTP groups (rows 3 and 5.5): MTP's draft heads, and GPU idle
% with leaders up to plain B = 128's idle and down to MTP B = 1's hatched idle.
\node[annot,fill=white,anchor=west] at (axis cs:1,4.1) {MTP draft heads};
\draw[leader] (axis cs:10.8,4.62) -- (axis cs:10.8,5.12);
\node[annot,fill=white,anchor=east] at (axis cs:93,4.1) {GPU idle};
\draw[leader] (axis cs:93.3,4.0) -- (axis cs:99.2,3.38) (axis cs:84,4.62) -- (axis cs:84,5.12);
% In the gap between the MTP and DFlash groups (rows 7.5 and 10): DFlash's draft heads.
\node[annot,fill=white,anchor=west] at (axis cs:1,8.6) {DFlash draft heads};
\draw[leader] (axis cs:7.2,9.12) -- (axis cs:7.2,9.62);
% group labels, rotated, left of the batch-size tick labels
\draw[ink!60] ([xshift=-0.98cm]axis cs:0,-0.3) -- ([xshift=-0.98cm]axis cs:0,3.3)
  ([xshift=-0.98cm]axis cs:0,5.2) -- ([xshift=-0.98cm]axis cs:0,7.8);
\node[annot,rotate=90,anchor=south] at ([xshift=-1.0cm]axis cs:0,1.5) {plain};
\node[annot,rotate=90,anchor=south] at ([xshift=-1.0cm]axis cs:0,6.5) {MTP};
\end{axis}
\end{tikzpicture}%
