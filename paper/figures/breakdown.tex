% Figure fig:breakdown: where the GPU time of a decode step goes. The bars read
% evidence/profiles/breakdown.csv (one row per configuration, columns in percent of the
% profiled step or cycle; the components sum to 100). Rows 0-3 of the file are plain
% decoding at B = 1, 8, 32, 128 and rows 4-6 the MTP cycle at B = 1, 8, 32 (column config);
% the plot leaves a gap of 0.6 rows between the two groups.
% Components: head = target head GEMM, FP32 logits copy and argmax; draft_head = the draft
% steps' head GEMMs, copies and top-1; gdn_state = the Gated DeltaNet recurrence and the
% other non-GEMM GDN kernels; other = attention, small kernels and the MTP layer's
% non-head work, its GEMMs included, so the legend calls weight_gemms "target GEMMs"
% (evidence/profiles/README.md). The head shares printed at the right are
% (head + draft_head) / step from the microseconds of evidence/profiles/step_share.csv,
% which round as the README's table does (the rounded percentages of breakdown.csv would
% give 12.5% instead of 12.4% for MTP at B = 32).
\vpreadcsv{\breakdowntable}{../evidence/profiles/breakdown.csv}
\vpreadcsv{\stepsharetable}{../evidence/profiles/step_share.csv}
\def\vpsplitconfig#1-#2\relax{\gdef\vparm{#1}\gdef\vpbatchsize{#2}}
\foreach \row in {0,...,6}{%
  \pgfplotstablegetelem{\row}{config}\of\breakdowntable
  \expandafter\vpsplitconfig\pgfplotsretval\relax
  \vplookupthree{\stepsharetable}{arm}{\vparm}{batch}{\vpbatchsize}{component}{head}{us_per_step}{\vphead}%
  \vplookupthree{\stepsharetable}{arm}{\vparm}{batch}{\vpbatchsize}{component}{draft_head}{us_per_step}{\vpdraft}%
  \vplookupthree{\stepsharetable}{arm}{\vparm}{batch}{\vpbatchsize}{component}{head}{step_us}{\vpstep}%
  \pgfmathsetmacro{\vpshare}{(\vphead+\vpdraft)/\vpstep*100}%
  \global\expandafter\let\csname vpheadshare\row\endcsname\vpshare}
\def\vpbatch#1-#2\relax{$B{=}#2$}
\begin{tikzpicture}
\begin{axis}[xbar stacked,y dir=reverse,scale only axis,width=5.9cm,height=1.9cm,
  bar width=3.9pt,xmin=0,xmax=100,ymin=-0.6,ymax=7.2,
  xlabel={Share of the profiled step or cycle},xlabel style={yshift=2pt},
  xtick={0,20,40,60,80,100},xticklabels={0\%,20\%,40\%,60\%,80\%,100\%},
  ytick=data,yticklabel={\pgfplotstablegetelem{\ticknum}{config}\of\breakdowntable
    \expandafter\vpbatch\pgfplotsretval\relax},
  major y tick style={draw=none},xmajorgrids,clip=false,
  every axis plot/.append style={draw=white,line width=.3pt},
  legend style={at={(-0.02,1.03)},anchor=south west,legend columns=3,fill=none,fill opacity=1,
    /tikz/every even column/.append style={column sep=3pt}},
  area legend]
\addplot[fill=ink] table[x=head,y expr={\thisrow{x}+0.6*(\thisrow{x}>3)},col sep=comma]{../evidence/profiles/breakdown.csv};
  \addlegendentry{head chain}
\addplot[fill=vpblue] table[x=draft_head,y expr={\thisrow{x}+0.6*(\thisrow{x}>3)},col sep=comma]{../evidence/profiles/breakdown.csv};
  \addlegendentry{draft heads}
\addplot[fill=ink!15] table[x=weight_gemms,y expr={\thisrow{x}+0.6*(\thisrow{x}>3)},col sep=comma]{../evidence/profiles/breakdown.csv};
  \addlegendentry{target GEMMs}
\addplot[fill=vpgreen!55] table[x=gdn_state,y expr={\thisrow{x}+0.6*(\thisrow{x}>3)},col sep=comma]{../evidence/profiles/breakdown.csv};
  \addlegendentry{GDN kernels}
\addplot[fill=ink!40] table[x expr={\thisrow{attention}+\thisrow{small_kernels}+\thisrow{draft_model}},y expr={\thisrow{x}+0.6*(\thisrow{x}>3)},col sep=comma]{../evidence/profiles/breakdown.csv};
  \addlegendentry{other kernels}
\addplot[fill=white,draw=muted,pattern=north east lines,pattern color=muted!80] table[x=idle,y expr={\thisrow{x}+0.6*(\thisrow{x}>3)},col sep=comma]{../evidence/profiles/breakdown.csv};
  \addlegendentry{GPU idle}
\pgfplotsinvokeforeach{0,...,6}{%
  \node[annot,anchor=west] at (axis cs:100.5,{#1+0.6*(#1>3)})
    {\pgfmathprintnumber[fixed,precision=1,fixed zerofill]{\csname vpheadshare#1\endcsname}\%};}
% group labels, rotated, left of the batch-size tick labels
\draw[muted] ([xshift=-0.98cm]axis cs:0,-0.3) -- ([xshift=-0.98cm]axis cs:0,3.3)
  ([xshift=-0.98cm]axis cs:0,4.3) -- ([xshift=-0.98cm]axis cs:0,6.9);
\node[annot,rotate=90,anchor=south] at ([xshift=-1.0cm]axis cs:0,1.5) {plain};
\node[annot,rotate=90,anchor=south] at ([xshift=-1.0cm]axis cs:0,5.6) {MTP};
\node[annot,anchor=south west] at (axis cs:100.5,-0.6) {head};
\end{axis}
\end{tikzpicture}
