% Figure fig:frontier: the confirmed serving frontier. Reads
% evidence/bench/confirm/frontier.csv (per arm and client concurrency: means over sessions
% of x = output tokens per second per request, end to end, and y = output tokens per second
% on the GPU; n = valid sessions) and evidence/bench/confirm/envelope-all.dat (the Pareto
% envelope over the points with at least three valid sessions, grey band). Points with fewer
% than three sessions are hollow; they do not rank on the frontier (evidence/bench/README.md). Drafter colours follow Figure
% fig:rho (DFlash ink, MTP purple); plain decoding is the grey baseline. Each curve runs
% from c = 1 (lower right) to c = 128 (upper left); the caption says so.
\pgfplotsset{only if n at least/.style={x filter/.append code={%
  \ifnum\thisrow{n}<#1\relax\def\pgfmathresult{inf}\fi}}}
\begin{tikzpicture}
\begin{axis}[xmode=log,ymode=log,width=\columnwidth,height=4.25cm,
  xmin=80,xmax=1150,ymin=220,ymax=18000,grid=major,
  xlabel={Per-request rate $x$ (output tokens/s)},
  ylabel={Throughput $y$ (output tokens/s)},
  xtick={100,200,500,1000},xticklabels={100,200,500,1000},
  ytick={300,1000,3000,10000},yticklabels={300,1k,3k,10k},
  every axis plot/.append style={mark size=1.6pt}]
\addplot[line width=5pt,ink!10,forget plot] table[x=x,y=y]{../evidence/bench/confirm/envelope-all.dat};
\addplot[thick,muted,mark=*,discard if not={label}{plain-tuned}] table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[thick,vppurple,mark=none,forget plot,discard if not={label}{mtp-tuned}] table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[only marks,vppurple,mark=triangle*,discard if not={label}{mtp-tuned},only if n at least=3] table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[only marks,vppurple,mark=triangle,forget plot,discard if not={label}{mtp-tuned}] table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[thick,ink,mark=diamond*,discard if not={label}{dflash-tuned}] table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[thick,ink,dashed,mark=square,mark options={solid},discard if not={label}{dflash-tuned-b16}] table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\node[annot,text=black,anchor=east] at (axis cs:255,600) {plain};
\node[annot,text=black,anchor=west] at (axis cs:475,500) {MTP, 3 steps};
\node[annot,text=black,anchor=south] at (axis cs:330,7600) {DFlash, block 8};
\node[annot,text=black,anchor=north west] at (axis cs:116,4400) {block 16};
\end{axis}
\end{tikzpicture}
