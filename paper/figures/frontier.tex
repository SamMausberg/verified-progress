% Figure fig:frontier: the confirmed serving frontier. Its width is the line width up to 10.5 cm
% and its height about half of that, so it keeps its shape in one column of the paper or across
% the one-column appendix (Appendix E). Reads
% evidence/bench/confirm/frontier.csv (per arm and client concurrency: means over sessions
% of x = output tokens per second per request, end to end, and y = output tokens per second
% on the GPU; n = valid sessions) and evidence/bench/confirm/envelope-all.dat (the Pareto
% envelope over the points with at least three valid sessions, grey band). Points with fewer
% than three sessions are hollow; they do not rank on the frontier (evidence/bench/README.md).
% Colours follow style.tex: plain decoding, the stock baseline, in ink; the drafter arms in
% purple, told apart by marker and dash pattern. Each curve runs from c = 1 (lower right) to
% c = 128 (upper left); the plain curve's ends are labelled. Error bars are one standard
% deviation over sessions (x_e2e_std, y_std); every one is below 2% of its mean and hidden by
% its marker, which the caption says. No arm uses the certified head.
% Keep only points with at least #1 sessions, or with fewer than #1.
\pgfplotsset{
  only if n at least/.style={x filter/.append code={%
    \ifnum\thisrow{n}<#1\relax\def\pgfmathresult{inf}\fi}},
  only if n below/.style={x filter/.append code={%
    \ifnum\thisrow{n}<#1\relax\else\def\pgfmathresult{inf}\fi}}}
\pgfmathsetmacro{\frontierw}{min(\linewidth,10.5cm)}
\pgfmathsetmacro{\frontierh}{0.515*\frontierw}
\begin{tikzpicture}
\begin{axis}[xmode=log,ymode=log,width=\frontierw pt,height=\frontierh pt,
  xmin=80,xmax=1150,ymin=220,ymax=18000,grid=major,
  xlabel={Per-request rate $x$ (output tokens/s)},
  ylabel={Throughput $y$ (output tokens/s)},
  xtick={100,200,500,1000},xticklabels={100,200,500,1000},
  ytick={300,1000,3000,10000},yticklabels={300,1k,3k,10k},
  every axis plot/.append style={mark size=1.6pt},
  % one standard deviation over sessions; keep last in a plot's options (it changes key path)
  sd bars/.style={error bars/.cd,x dir=both,x explicit,y dir=both,y explicit,error bar style={thin}}]
\addplot[line width=4pt,ccontext!25,forget plot,no markers]
  table[x=x,y=y]{../evidence/bench/confirm/envelope-all.dat};
\addplot[cstock,thick,mark=*,discard if not={label}{plain-tuned},sd bars]
  table[x=x_e2e_mean,y=y_mean,x error=x_e2e_std,y error=y_std,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[cdraft,thick,dashdotted,mark=none,forget plot,discard if not={label}{mtp-tuned}]
  table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[cdraft,only marks,mark=triangle*,mark size=1.9pt,discard if not={label}{mtp-tuned},only if n at least=3,sd bars]
  table[x=x_e2e_mean,y=y_mean,x error=x_e2e_std,y error=y_std,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[cdraft,only marks,mark=triangle,mark size=1.9pt,forget plot,discard if not={label}{mtp-tuned},only if n below=3]
  table[x=x_e2e_mean,y=y_mean,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[cdraft,thick,mark=diamond*,mark size=1.9pt,discard if not={label}{dflash-tuned},sd bars]
  table[x=x_e2e_mean,y=y_mean,x error=x_e2e_std,y error=y_std,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\addplot[cdraft,thick,dashed,mark=square*,mark size=1.4pt,mark options={solid},discard if not={label}{dflash-tuned-b16},sd bars]
  table[x=x_e2e_mean,y=y_mean,x error=x_e2e_std,y error=y_std,col sep=comma]{../evidence/bench/confirm/frontier.csv};
\node[annot,text=black,anchor=east] at (axis cs:255,600) {plain};
\node[annot,text=black,anchor=west] at (axis cs:475,500) {MTP, 3 steps};
\node[annot,text=black,anchor=south] at (axis cs:330,7600) {DFlash, block 8};
\node[annot,text=black,anchor=north west] at (axis cs:116,4400) {block 16};
\node[annot,text=black,anchor=west] at (axis cs:300,282) {$c{=}1$};
\node[annot,text=black,anchor=west] at (axis cs:112,15500) {$c{=}128$};
\node[annot,text=black,anchor=south west,align=left] at (axis cs:84,240) {stock arms, measured;\\no certified head};
\end{axis}
\end{tikzpicture}
