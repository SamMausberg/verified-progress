% Figure fig:rho. (a) is drawn to scale from medians read from committed files:
% ||h^t||_2 and ||h^d||_2 from evidence/head_geometry/transport_dflash4b.csv (metrics htnorm2
% and hdnorm2, group all, column median), the threshold delta* from
% evidence/head_geometry/rho_dflash4b_quantiles.csv (threshold_int8_row at q = 0.5), and
% cos(h^d, h^t) = 0.40811888453704215 from evidence/head_geometry/stats_dflash4b.json
% (pairs/cos_hd_ht/all/q50), which exists only in JSON and is copied here. Every number
% printed in (a) is computed from these four values. (b) reads
% evidence/head_geometry/rho_dflash4b_quantiles.csv (DFlash-4B drift and the per-row
% thresholds) and evidence/head_geometry/rho_mtp4b_quantiles.csv (MTP-4B drift); both
% hold 40,000 held-out pairs.
% Both panels are one picture: (b)'s plot area is centred on y = 0, (a)'s drawing sits
% beside it below the shared title line, and (a)'s title is set on (b)'s title line.
\vpreadcsv{\transporttable}{../evidence/head_geometry/transport_dflash4b.csv}
\vpreadcsv{\rhotable}{../evidence/head_geometry/rho_dflash4b_quantiles.csv}
\vplookuptwo{\transporttable}{metric}{htnorm2}{group}{all}{median}{\figHt}
\vplookuptwo{\transporttable}{metric}{hdnorm2}{group}{all}{median}{\figHd}
\vplookup{\rhotable}{q}{0.5}{threshold_int8_row}{\figDstar}
\vplookup{\rhotable}{q}{0.5}{rho_all}{\figRho}
\pgfmathsetmacro{\figCos}{0.40811888453704215}
\pgfmathsetmacro{\figSin}{sqrt(1-\figCos*\figCos)}
\pgfmathsetmacro{\figNormRatio}{\figHd/\figHt}
\pgfmathsetmacro{\figDrel}{sqrt(1+\figNormRatio*\figNormRatio-2*\figNormRatio*\figCos)}
\begin{tikzpicture}[font=\footnotesize]
% ---------------- (b) measured drift against the thresholds ----------------
\begin{semilogxaxis}[name=b,at={(11.2cm,0)},anchor=center,
  width=8.7cm,height=4.15cm,xmin=1e-3,xmax=1.6,ymin=0,ymax=1,
  xlabel={Drift ratio $\rho$ of a pair, or per-row threshold $\delta^\ast_i$},ylabel={Cumulative share},grid=major,
  ytick={0,0.25,0.5,0.75,1},yticklabels={0,0.25,0.5,0.75,1},
  title={\paneltitle{b}{Measured drift against derived thresholds}},xminorticks=false]
\addplot[series int8 g128] table[x=threshold_int8_g128,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
\addplot[series int8 row] table[x=threshold_int8_row,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
\addplot[series int4] table[x=threshold_int4_row,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
\addplot[series dflash] table[x=rho_all,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
\addplot[series mtp,mark repeat=25,mark phase=13] table[x=rho_all,y=q,col sep=comma]{../evidence/head_geometry/rho_mtp4b_quantiles.csv};
\draw[{Stealth[length=1.6mm]}-{Stealth[length=1.6mm]},muted] (axis cs:\figDstar,0.5) -- (axis cs:\figRho,0.5);
\node[annot,fill=white] at (axis cs:0.037,0.5) {medians $\vpnum{0}{\figRho/\figDstar}\times$ apart};
\node[annot,anchor=east,align=right] at (axis cs:0.0052,0.3) {$\delta^\ast_i$: int8,\\128 cols};
\node[annot,anchor=west] at (axis cs:0.0084,0.3) {$\delta^\ast_i$: int8, per row};
\node[annot,anchor=east,align=right] at (axis cs:0.155,0.8) {$\delta^\ast_i$: int4,\\per row};
\node[annot,anchor=east] at (axis cs:0.9,0.8) {$\rho$: DFlash-4B};
\node[annot,anchor=east] at (axis cs:0.74,0.1) {$\rho$: MTP-4B};
\end{semilogxaxis}
