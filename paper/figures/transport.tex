% Figure fig:rho. (a) is drawn to scale from medians read from committed files:
% ||h^t||_2 and ||h^d||_2 from evidence/head_geometry/transport_dflash4b.csv (metrics htnorm2
% and hdnorm2, group all, column median; 4,020 pairs), the threshold delta* from
% evidence/head_geometry/rho_dflash4b_quantiles.csv (threshold_int8_row at q = 0.5), and
% cos(h^d, h^t) from evidence/head_geometry/stats_dflash4b.json (pairs/cos_hd_ht/all/q50), which
% exists only in JSON and is read from figures/data/json_values.csv. Every number printed in (a)
% is computed from these four values. Colours: h^t ink (the target), h^d and Delta purple (the
% drafter and its drift), the ball blue (the int8 copy's threshold). (b) reads
% evidence/head_geometry/rho_dflash4b_quantiles.csv (DFlash-4B drift and the per-row
% thresholds) and evidence/head_geometry/rho_mtp4b_quantiles.csv (MTP-4B drift); both
% hold 40,000 held-out pairs. Thresholds are derived from the weights (lines without markers);
% the drift is measured (sparse markers).
% Both panels are one picture: (b)'s plot area is centred on y = 0, (a)'s drawing sits
% beside it below the shared title line, and (a)'s title is set on (b)'s title line.
\vpreadcsv{\transporttable}{../evidence/head_geometry/transport_dflash4b.csv}
\vpreadcsv{\rhotable}{../evidence/head_geometry/rho_dflash4b_quantiles.csv}
\vplookuptwo{\transporttable}{metric}{htnorm2}{group}{all}{median}{\figHt}
\vplookuptwo{\transporttable}{metric}{hdnorm2}{group}{all}{median}{\figHd}
\vplookup{\rhotable}{q}{0.5}{threshold_int8_row}{\figDstar}
\vplookup{\rhotable}{q}{0.5}{rho_all}{\figRho}
\vpjson{dflash4b_cos_hd_ht_median}{\figCosRaw}
\pgfmathsetmacro{\figCos}{\figCosRaw}
\pgfmathsetmacro{\figSin}{sqrt(1-\figCos*\figCos)}
\pgfmathsetmacro{\figNormRatio}{\figHd/\figHt}
\pgfmathsetmacro{\figDrel}{sqrt(1+\figNormRatio*\figNormRatio-2*\figNormRatio*\figCos)}
\begin{tikzpicture}[font=\footnotesize]
% ---------------- (b) measured drift against the thresholds ----------------
\begin{semilogxaxis}[name=b,at={(11.2cm,0)},anchor=center,
  width=8.7cm,height=3.95cm,xmin=1e-3,xmax=20,ymin=0,ymax=1,
  % a key for the two drafters in the empty upper right (both curves lie left of 1.4)
  legend style={at={(axis cs:19,0.97)},anchor=north east},
  legend image code/.code={\draw[mark repeat=2,mark phase=2,#1] plot coordinates {(0cm,0cm) (0.25cm,0cm) (0.5cm,0cm)};},
  xlabel={Drift ratio $\rho$ of a pair, or threshold $\delta^\ast_i$ of an entry},ylabel={Cumulative share},grid=major,
  ytick={0,0.25,0.5,0.75,1},yticklabels={0,0.25,0.5,0.75,1},
  title={\paneltitle{b}{Measured drift against derived thresholds}},xminorticks=false]
\addplot[copy int8 g128,derived,forget plot] table[x=threshold_int8_g128,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
\addplot[copy int8 row,derived,forget plot] table[x=threshold_int8_row,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
\addplot[copy int4,derived,forget plot] table[x=threshold_int4_row,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
% DFlash: solid, filled circles; MTP: lighter purple, dashed, hollow triangles
\addplot[drafter dflash,mark=*,mark size=1.6pt,mark repeat=50,mark phase=1] table[x=rho_all,y=q,col sep=comma]{../evidence/head_geometry/rho_dflash4b_quantiles.csv};
\addlegendentry{DFlash-4B}
\addplot[cdraft!60,thick,dash pattern=on 3.5pt off 1.8pt,mark=triangle*,mark size=2.1pt,mark options={solid,fill=white,draw=cdraft},mark repeat=50,mark phase=26] table[x=rho_all,y=q,col sep=comma]{../evidence/head_geometry/rho_mtp4b_quantiles.csv};
\addlegendentry{MTP-4B}
\draw[{Stealth[length=1.6mm]}-{Stealth[length=1.6mm]},ink!70] (axis cs:\figDstar,0.5) -- (axis cs:\figRho,0.5);
% above the arrow, between the int8 and int4 threshold curves
\node[annot,anchor=south,align=center] at (axis cs:0.025,0.515) {medians\\$\vpnum{0}{\figRho/\figDstar}\times$ apart};
\node[annot,anchor=east,align=right] at (axis cs:0.0052,0.3) {$\delta^\ast_i$:\\int8, 128\\columns};
\node[annot,anchor=west] at (axis cs:0.0084,0.3) {$\delta^\ast_i$: int8, per entry};
\node[annot,anchor=east,align=right] at (axis cs:0.155,0.87) {$\delta^\ast_i$: int4,\\per entry};
\end{semilogxaxis}
% ---------------- (a) median DFlash-4B head inputs, to scale ----------------
% The drawing spans y from about -41 units (the callout text) to 62 (the line through h^d);
% it is shifted down 17 units so that its top stays below the shared title line.
\begin{scope}[x=0.0295cm,y=0.0295cm,shift={(0,-17)}]
\coordinate (O) at (0,0);
\coordinate (T) at (\figHt,0);
\coordinate (D) at ({\figHd*\figCos},{\figHd*\figSin});

\coordinate (amid) at (68,0);

% the tip of h^t; F is marked, and the dotted perpendicular has length rho = sin(angle) ||h^t||
\draw[cdraft,densely dotted] (D) -- (F);
\draw[ink!60,densely dotted] (T) -- (F);
\fill[cdraft] (F) circle[radius=1.1pt];
\node[note,anchor=south west,align=left] at ($(T)!.4!(F)+(2,2)$) {closest rescaling of $h^d$:\\$\rho=\vpnum{2}{\figSin}$};

\draw[flow,very thick,shorten >=2.2pt] (O) -- (T);

\node[anchor=south east,align=right,inner sep=1pt] at ($(D)+(-1,1)$) {$h^d$, $\norm{h^d}_2=\vpnum{1}{\figHd}$};

\node[panel title,anchor=north] at (amid |- b.outer north) {\paneltitle{a}{Median DFlash-4B head inputs, to scale}};
