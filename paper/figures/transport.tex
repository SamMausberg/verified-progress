% Figure fig:rho. (a) is drawn to scale from medians read from committed files:
% ||h^t||_2 and ||h^d||_2 from evidence/head_geometry/transport_dflash4b.csv (metrics htnorm2
% and hdnorm2, group all, column median; 4,020 pairs), the threshold delta* from
% evidence/head_geometry/rho_dflash4b_quantiles.csv (threshold_int8_row at q = 0.5), and
% cos(h^d, h^t) from evidence/head_geometry/stats_dflash4b.json (pairs/cos_hd_ht/all/q50), which
% exists only in JSON and is read from figures/data/json_values.csv. Every number printed in (a)
% is computed from these four values. Colours: h^t ink (the target), h^d and Delta purple (the
% drafter and its drift), the ball blue (the int8 copy's threshold). (b) reads
% evidence/head_geometry/rho_dflash4b_quantiles.csv (DFlash-4B drift and the per-entry
% thresholds) and (c) evidence/head_geometry/rho_mtp4b_quantiles.csv (MTP-4B drift) with the same
% thresholds; both hold 40,000 held-out pairs. Thresholds are derived from the weights (lines
% without markers); the drift is measured (sparse markers). The printed ratios are median drift
% over median int8 per-entry threshold (107 for DFlash-4B, 112 for MTP-4B).
% All panels are one picture: (b) and (c) are stacked on the right with a shared x axis, (a)'s
% drawing sits beside them below the shared title line, and (a)'s title is set on (b)'s.
\vpreadcsv{\transporttable}{../evidence/head_geometry/transport_dflash4b.csv}
\vpreadcsv{\rhotable}{../evidence/head_geometry/rho_dflash4b_quantiles.csv}
\vplookuptwo{\transporttable}{metric}{htnorm2}{group}{all}{median}{\figHt}
\vplookuptwo{\transporttable}{metric}{hdnorm2}{group}{all}{median}{\figHd}
\vplookup{\rhotable}{q}{0.5}{threshold_int8_row}{\figDstar}
\vplookup{\rhotable}{q}{0.5}{rho_all}{\figRho}
\vpjson{dflash4b_cos_hd_ht_median}{\figCosRaw}
\pgfmathsetmacro{\figCos}{\figCosRaw}
\pgfmathsetmacro{\figSin}{sqrt(1-\figCos*\figCos)}
\pgfmathsetmacro{\figNormRatio}{\figHd/\figHt}
\pgfmathsetmacro{\figDrel}{sqrt(1+\figNormRatio*\figNormRatio-2*\figNormRatio*\figCos)}
\vpreadcsv{\rhomtptable}{../evidence/head_geometry/rho_mtp4b_quantiles.csv}
\vplookup{\rhomtptable}{q}{0.5}{rho_all}{\figRhoMtp}
\def\rhodflash{../evidence/head_geometry/rho_dflash4b_quantiles.csv}
\def\rhomtp{../evidence/head_geometry/rho_mtp4b_quantiles.csv}
\begin{tikzpicture}[font=\figlabel]
% ---------------- (b) DFlash-4B and (c) MTP-4B: measured drift against the thresholds -------
% Two stacked panels (style panel stacked, 1.9 cm), one drafter each, on a shared x axis; both
% show the same derived thresholds (lines without markers), labelled in (b) in regions that no
% curve enters (checked against the quantile files: the per-128-weights threshold is above 0.0061
% for q > 0.52, the per-entry one below 0.0083 for q < 0.4, int4's above 0.12, DFlash-4B's drift
% above 0.85 and MTP-4B's above 0.8 for q > 0.15). The arrow joins the median threshold (int8,
% per entry) to the drafter's median drift. Drifts use the arm styles: DFlash diamonds, MTP
% triangles, as in Figure fig:frontier. The x axis starts at 4e-4, not 1e-3, so that the
% per-128-weights label fits in two lines left of its curve.
\pgfplotsset{drift panel/.style={panel stacked,width=7.1cm,xmin=4e-4,xmax=10,
  ymin=0,ymax=1,ytick={0,0.5,1},yticklabels={0,0.5,1},grid=major,xminorticks=false}}
\begin{semilogxaxis}[drift panel,name=b,at={(7.85cm,0.25cm)},anchor=south west,xticklabels={},
  title={\paneltitle{b}{DFlash-4B: measured drift, derived thresholds}}]
\addplot[copy int8 g128,derived] table[x=threshold_int8_g128,y=q,col sep=comma]{\rhodflash};
\addplot[copy int8 row,derived] table[x=threshold_int8_row,y=q,col sep=comma]{\rhodflash};
\addplot[copy int4,derived] table[x=threshold_int4_row,y=q,col sep=comma]{\rhodflash};
\addplot[arm dflash,mark repeat=50,mark phase=25] table[x=rho_all,y=q,col sep=comma]{\rhodflash};
\draw[{Stealth[length=1.5mm]}-{Stealth[length=1.5mm]},leader] (axis cs:\figDstar,0.5) -- (axis cs:\figRho,0.5);

\node[annot,anchor=north,align=center] at (axis cs:0.42,0.45) {medians\\$\vpnum{0}{\figRho/\figDstar}\times$};
\node[annot,anchor=north east,align=right] at (axis cs:0.0055,0.85) {$\delta^\ast_i$: int8,\\per 128 weights};
\node[annot,anchor=west,align=left] at (axis cs:0.0098,0.25) {$\delta^\ast_i$: int8,\\per entry};
\node[annot,anchor=east,align=right] at (axis cs:0.10,0.8) {$\delta^\ast_i$: int4,\\per entry};
\node[annot,anchor=west] at (axis cs:1.1,0.3) {$\rho$, DFlash-4B};
\end{semilogxaxis}
\begin{semilogxaxis}[drift panel,name=c,at={(7.85cm,-2.25cm)},anchor=south west,

  title={\paneltitle{c}{MTP-4B: measured drift, same thresholds}}]
\addplot[copy int8 g128,derived] table[x=threshold_int8_g128,y=q,col sep=comma]{\rhodflash};
\addplot[copy int8 row,derived] table[x=threshold_int8_row,y=q,col sep=comma]{\rhodflash};
\addplot[copy int4,derived] table[x=threshold_int4_row,y=q,col sep=comma]{\rhodflash};

\node[annot,anchor=north,align=center] at (axis cs:0.38,0.45) {medians\\$\vpnum{0}{\figRhoMtp/\figDstar}\times$};
\node[annot,anchor=west] at (axis cs:0.02,0.85) {$\delta^\ast_i$ as in (b)};
\node[annot,anchor=west] at (axis cs:1.1,0.3) {$\rho$, MTP-4B};
\end{semilogxaxis}
% one y label for both panels
